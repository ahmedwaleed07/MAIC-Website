مثال لبنية السؤال في ملف JSON

{
  "format": "maic-mcq-v1",
  "questions": [
    {
      "stem": "Which planning practice best supports scope control?",
      "options": ["A", "B", "C", "D"],
      "correct": [0, 2],
      "page": 5,
      "rationale": "Explain precisely why these options follow the approved course material."
    }
  ]
}
